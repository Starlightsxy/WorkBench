	<script setup>
	import { ApiResponse, UserInfoVO } from '../../type';
	import { getTokenStorage, getUserStorage, setUserStorage } from '../../storage'
	// 在页面加载的时候执行一些操作
	onLoad(() => {
		// 获取用户信息并存储在本地
		getUser()
	})
	// 从本地中获取用户信息
	const userInfo = getUserStorage()
	// 获取时间
	const now = new Date()
	const dayStr = ['一', '二', '三', '四', '五', '六', '七']

	// 判断当前时间应该是那个
	const timeGreeting = computed(() => {
		const nowHourse = now.getHours()
		if (nowHourse >= 0 && nowHourse < 6) return "凌晨好：";
		if (nowHourse >= 6 && nowHourse < 11) return "早上好：";
		if (nowHourse >= 11 && nowHourse < 13) return "中午好：";
		if (nowHourse >= 13 && nowHourse < 18) return "下午好：";
		if (nowHourse >= 18 && nowHourse < 24) return "晚上好：";
	})
	// 当前月份和周几
	const nowDay = computed(() => {
		return `${now.getMonth() + 1}月${now.getDate()}日 周${dayStr[now.getDay() - 1]}`
	})
	// 获取用户信息并存储在本地
	const getUser = () => {
		uni.request({
			url: "http://localhost:3000/user/getUserInfo",
			method: "GET",
			header: {
				Authorization: 'Bearer ' + getTokenStorage()
			},
			success: (e : RequestSuccess<ApiResponse<UserInfoVO>>) => {
				// 失败
				if (e.statusCode !== 200) {
					uni.showToast({
						title: e.data + ""
					})
					return
				}
				// 成功
				if (e.data?.code === 1) {
					//
					setUserStorage(e.data.data)
				}


			}
		})
	}
	</script>
	<template>
		<view style=" box-sizing: border-box; border: 1px red solid;background-color:#f5f0f4; width: 100%; height:calc(100vh - 100px);">
			<view class="flex-space-between">
				<view class="fts-12 pd-5">首页</view>
				<view>
					<view class="fts-12 pd-5">{{nowDay}}</view>
				</view>
			</view>
			<view class="txt-center fts-14">{{timeGreeting +userInfo.nickName}}</view>
			<view class="txt-center" style="font-size: 50px;">
				<!-- 当分钟是0-9区间时使用 String(now.getMinutes()).padStart(2, '0') 补0 -->
				{{now.getHours() +":" + String(now.getMinutes()).padStart(2, '0')}}
			</view>
			<view class="txt-center fts-14">
				{{nowDay}}
			</view>
			<view class="txt-center fts-14 mt-10" style="background-color:#faf5f9 ; border:1px #daced6 solid ;border-radius: 10px; width: 80%; margin-left: 10%; height: 13%; justify-content: center;">
				{{`"${userInfo.signature}"`}}


			</view>
		</view>
	</template>
